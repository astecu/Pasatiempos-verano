create or replace function public.generate_share_code()
returns text
language sql
volatile
as $$
  select upper(substr(replace(gen_random_uuid()::text, '-', ''), 1, 8));
$$;

-- Prueba opcional: debe devolver un código de 8 caracteres
select public.generate_share_code() as codigo_prueba;
